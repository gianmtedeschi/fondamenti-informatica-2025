\begin{Exercise}[label={ex:c2}]
    Utilizzando 8 bit, rappresentare il numero decimale $-42_{10}$ in Complemento a 2 (C2).
\end{Exercise}

\begin{Answer}[ref={ex:c2}]
    \textbf{1. Binario del Valore Assoluto (+42 su 8 bit):}
    \[ +42_{10} = 00101010 \]
    \textbf{2. Complemento a 1 (Inversione dei bit):}
    \[ 00101010 \quad \to \quad 11010101 \]
    \textbf{3. Complemento a 2 (Aggiungere 1):}
    \begin{align*}
        11010101 \\
        + \quad 1 \\
        \hline
        \mathbf{11010110}
    \end{align*}

    Risultato: $\mathbf{11010110_{C2}}$
\end{Answer}
